cmake_minimum_required(VERSION 3.10)
project(HighZoneDetector)
set(CMAKE_CXX_STANDARD 17)
find_package(OpenCV REQUIRED)
add_executable(high_zone_detector main.cpp)
target_include_directories(high_zone_detector PRIVATE ${OpenCV_INCLUDE_DIRS})
target_link_libraries(high_zone_detector PRIVATE ${OpenCV_LIBS})
